# Zigbee-Funkmodul (RaspBee II auf der GPIO-Leiste oder ConBee am USB):
#   auto = beim Start erkennen; deCONZ/Phoscon nur starten, wenn ein Modul antwortet
#   on   = deCONZ immer starten (falls die Erkennung das Modul nicht findet)
#   off  = deCONZ nie starten
ZIGBEE=auto

# Gerät erzwingen (leer = automatisch; RaspBee II: /dev/ttyAMA0)
#ZIGBEE_DEVICE=/dev/ttyAMA0
